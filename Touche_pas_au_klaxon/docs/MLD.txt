AGENCIES(
  #id INT UNSIGNED AUTO_INCREMENT,
  name VARCHAR(100) UNIQUE NOT NULL
)

USERS(
  #id INT UNSIGNED AUTO_INCREMENT,
  last_name VARCHAR(100) NOT NULL,
  first_name VARCHAR(100) NOT NULL,
  phone VARCHAR(20) NOT NULL,
  email VARCHAR(190) UNIQUE NOT NULL,
  password_hash VARCHAR(255) NOT NULL,
  role ENUM('user','admin') NOT NULL,
  created_at DATETIME NOT NULL
)

TRIPS(
  #id INT UNSIGNED AUTO_INCREMENT,
  departure_at DATETIME NOT NULL,
  arrival_at DATETIME NOT NULL,
  seats_total TINYINT UNSIGNED NOT NULL,
  seats_available TINYINT UNSIGNED NOT NULL,
  created_at DATETIME NOT NULL,
  updated_at DATETIME NOT NULL,
  departure_agency_id INT UNSIGNED NOT NULL => AGENCIES.id,
  arrival_agency_id INT UNSIGNED NOT NULL => AGENCIES.id,
  owner_id INT UNSIGNED NOT NULL => USERS.id
)

Cardinalités :
- Un utilisateur propose 0..N trajets ; chaque trajet a exactement 1 auteur (UTILISATEUR).
- Une agence accueille 0..N départs ; chaque trajet a exactement 1 agence de départ (AGENCE).
- Une agence accueille 0..N arrivées ; chaque trajet a exactement 1 agence d'arrivée (AGENCE).

Règles : départ != arrivée ; arrivée_at > départ_at ; 1 <= sièges_total <= 8 ;
0 <= sièges_disponibles <= sièges_total. L'interface publique n'affiche que
les trajets de départ futur avec sièges_disponibles > 0, par date ascendante.

Les employés proviennent uniquement du jeu RH : aucun CRUD salarié via le site.
